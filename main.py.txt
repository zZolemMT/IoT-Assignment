import asyncio
import time
from machine import Pin
import buzzer.py

LEFT_BUTTON = Pin(6, Pin.IN, Pin.PULL_UP)
RIGHT_BUTTON = Pin(7, Pin.IN, Pin.PULL_UP)
BUZZER_PIN = Pin(10)

# Note format: (seconds_from_start, lane, note_name)
# lane 0 = left button, lane 1 = right button
PATTERN = [
    (0.55, 0, "3"),
    (1.15, 1, "5"),
    (1.75, 0, "4"),
    (2.35, 1, "5"),
    (2.95, 0, "3"),
    (3.55, 1, "2"),
    (4.15, 0, "1"),
    (4.75, 1, "3"),
    (5.35, 0, "3"),
    (5.95, 1, "5"),
    (6.55, 0, "4"),
    (7.15, 1, "3"),
]

TARGET_WINDOW_MS = 150
NOTE_TEMPO = 150

beeper = buzzer.Buzzer(BUZZER_PIN)


async def play_note(note_name):
    await beeper.async_play([(1, note_name)], tempo=NOTE_TEMPO)


async def rhythm_game():
    start_ms = time.ticks_ms()
    index = 0
    score = 0
    combo = 0

    left_last = LEFT_BUTTON.value() == 0
    right_last = RIGHT_BUTTON.value() == 0

    print("Rhythm game started")

    while index < len(PATTERN):
        now_ms = time.ticks_ms() - start_ms
        target_ms = int(PATTERN[index][0] * 1000)

        if now_ms > target_ms + TARGET_WINDOW_MS:
            combo = 0
            print("MISS")
            index += 1
            await asyncio.sleep_ms(10)
            continue

        left_now = LEFT_BUTTON.value() == 0
        right_now = RIGHT_BUTTON.value() == 0

        if left_now and not left_last:
            if PATTERN[index][1] == 0 and abs(now_ms - target_ms) <= TARGET_WINDOW_MS:
                score += 100 + combo * 10
                combo += 1
                print("HIT", score)
                loop = asyncio.get_event_loop()
                loop.create_task(play_note(PATTERN[index][2]))
                index += 1
            else:
                combo = 0
                score = max(0, score - 25)
                print("BAD", score)

        elif right_now and not right_last:
            if PATTERN[index][1] == 1 and abs(now_ms - target_ms) <= TARGET_WINDOW_MS:
                score += 100 + combo * 10
                combo += 1
                print("HIT", score)
                loop = asyncio.get_event_loop()
                loop.create_task(play_note(PATTERN[index][2]))
                index += 1
            else:
                combo = 0
                score = max(0, score - 25)
                print("BAD", score)

        left_last = left_now
        right_last = right_now
        await asyncio.sleep_ms(5)

    print("Game over! Final score:", score)


loop = asyncio.get_event_loop()
loop.run_until_complete(rhythm_game())
