from machine import Pin, PWM
import time

# led = Pin(1, Pin.OUT)
buzzer = PWM(Pin(0)) 
notes = [262, 294, 330] 

while True:
    for note in notes:
        buzzer.freq(note)
        buzzer.duty_u16(10000)
        time.sleep(0.2)  # Duration of each note

buzzer.duty_u16(0)
buzzer.deinit()
