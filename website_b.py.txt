from machine import Pin
import dht
import network
import socket
import time
import rp2

button = Pin(15, Pin.IN, Pin.PULL_UP)
sensor = dht.DHT11(Pin(13))

rp2.country('AU')
ssid = "iPhone"
password = 'password'

wlan = network.WLAN(network.STA_IF)
wlan.active(True)
for w in wlan.scan():
    print(w)

wlan.connect(ssid, password)

max_wait = 20
while max_wait > 0:
    if wlan.status() >= 3:
        break
    max_wait -= 1
    time.sleep(1)

if wlan.status() != 3:
    raise RuntimeError('Network connection failed')

print("Connected! IP:", wlan.ifconfig()[0])

addr = socket.getaddrinfo('0.0.0.0', 80)[0][-1]
previous_socket = globals().get('s')
if previous_socket is not None:
    previous_socket.close()
s = socket.socket()
s.bind(addr)
s.listen(1)
print('Listening on', addr)
print(wlan.status())
print(wlan.ifconfig())

html = """<!doctype html>
<html>
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Button Status</title>
</head>
<body>
    <h1 id="status">Checking button...</h1>
    <script>
        async function updateStatus() {
            try {
                const response = await fetch('/state', { cache: 'no-store' });
                document.getElementById('status').textContent = await response.text();
            } catch (error) {
                document.getElementById('status').textContent = 'Unable to get button status';
            }
        }
        updateStatus();
        setInterval(updateStatus, 500);
    </script>
</body>
</html>
"""

while True:
    cl = None

    try:
        cl, addr = s.accept()
        print("Client connected:", addr)

        request = cl.recv(1024)
        print("Request received")

        request_line = request.split(b"\r\n", 1)[0]
        if b"GET /state " in request_line:
            body = "Button Pressed!" if button.value() == 0 else "Button Not Pressed"
            content_type = "text/plain; charset=UTF-8"
        else:
            body = html
            content_type = "text/html; charset=UTF-8"

        body_bytes = body.encode("utf-8")
        response = (
            "HTTP/1.1 200 OK\r\n"
            "Content-Type: " + content_type + "\r\n"
            "Content-Length: " + str(len(body_bytes)) + "\r\n"
            "Cache-Control: no-store\r\n"
            "Connection: close\r\n"
            "\r\n"
        ).encode("utf-8") + body_bytes

        cl.sendall(response)

        print("Page sent")

    except Exception as e:
        print("Server error:", e)

    finally:
        if cl:
            cl.close()
