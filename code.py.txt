import time
import board
import digitalio
import RPi.GPIO as GPIO

print("Starting keyboard test")

try:
    import usb_hid
    keyboard = next(
        device
        for device in usb_hid.devices
        if device.usage_page == 0x01 and device.usage == 0x06
    )
except Exception as error:
    print("USB keyboard setup failed:", error)
    while True:
        time.sleep(1)

button = digitalio.DigitalInOut(board.GP15)
button.direction = digitalio.Direction.INPUT
button.pull = digitalio.Pull.UP  # Uses internal resistor; connects to GND when pressed

was_pressed = False
last_status = time.monotonic()
print("Ready: GP15 to GND to type SPACE")

buzzer_pin = 10

while True:
    is_pressed = not button.value
    if is_pressed and not was_pressed:
        keyboard.send_report(bytes((0, 0, 0x2C, 0, 0, 0, 0, 0)), report_id=1)
    elif not is_pressed and was_pressed:
        keyboard.send_report(bytes(8), report_id=1)
    was_pressed = is_pressed
    if time.monotonic() - last_status >= 2:
        last_status = time.monotonic()
