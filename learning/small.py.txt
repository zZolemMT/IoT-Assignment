import network
import socket
import time
import rp2

rp2.country('AU')

# Wi-Fi Credentials
ssid = "iPhone"
password = 'password'

wlan = network.WLAN(network.STA_IF)
wlan.active(True)
for w in wlan.scan():
    print(w)

# Connect to Wi-Fi
wlan.connect(ssid, password)

# Wait for connection
max_wait = 20
while max_wait > 0:
    if wlan.status() >= 3:
        break
    max_wait -= 1
    time.sleep(1)

if wlan.status() != 3:
    raise RuntimeError('Network connection failed')

print("Connected! IP:", wlan.ifconfig()[0])

# Create Socket
addr = socket.getaddrinfo("0.0.0.0", 80)[0][-1]

s = socket.socket()
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.bind(addr)
s.listen(1)

print("SERVER READY")
print("Open: http://" + wlan.ifconfig()[0])

while True:
    cl = None

    try:
        print("Waiting for connection...")
        cl, addr = s.accept()

        print("CONNECTED:", addr)

        request = cl.recv(1024)
        print("REQUEST:", request)

        response = """HTTP/1.0 200 OK
Content-Type: text/html
Connection: close

<html>
<body>
<h1>HELLO FROM PICO</h1>
<p>If you can see this, the server works.</p>
</body>
</html>
"""

        cl.sendall(response.encode())
        print("SENT!")

    except Exception as e:
        print("ERROR:", e)

    finally:
        if cl is not None:
            cl.close()
            print("CLOSED")
