from machine import Pin, PWM
import time

button = Pin(0, Pin.IN, Pin.PULL_UP)
buzzer = PWM(Pin(16))
notes = [2000, 1100]
# notes = [2620, 1310]

while True:
    if button.value() != 1:
        for note in notes:
            buzzer.freq(note)
            buzzer.duty_u16(16384)
            time.sleep(0.5)
    else:
        buzzer.duty_u16(0)
