import network
import socket
import time
import rp2

rp2.country('AU')

# Wi-Fi Credentials
ssid = "iPhone"
password = 'password'

wlan = network.WLAN(network.STA_IF)
wlan.active(True)
for w in wlan.scan():
    print(w)

# Connect to Wi-Fi
wlan.connect(ssid, password)

# Wait for connection
max_wait = 20
while max_wait > 0:
    if wlan.status() >= 3:
        break
    max_wait -= 1
    time.sleep(1)

if wlan.status() != 3:
    raise RuntimeError('Network connection failed')

print("Connected! IP:", wlan.ifconfig()[0])

# Create Socket
addr = socket.getaddrinfo('0.0.0.0', 80)[0][-1]
s = socket.socket()
s.bind(addr)
s.listen(1)
print('Listening on', addr)
print(wlan.status())
print(wlan.ifconfig())

# HTML Content
html = "Hello, World!"

# Serve Web Page
#hile True:

#       cl.send(html)

# Serve Web Page
while True:
    cl = None

    try:
        cl, addr = s.accept()
        print("Client connected:", addr)

        request = cl.recv(1024)
        print("Request received")

        response = (
            "HTTP/1.1 200 OK\r\n"
            "Content-Type: text/html; charset=UTF-8\r\n"
            "Connection: close\r\n"
            "\r\n"
            + html
        )

        cl.sendall(response.encode("utf-8"))

        print("Page sent")

    except Exception as e:
        print("Server error:", e)

    finally:
        if cl:
            cl.close()
